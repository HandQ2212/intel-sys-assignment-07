\section{Các kỹ thuật Học máy Cơ bản (Machine Learning)}

\subsection{Giới thiệu chung về các kỹ thuật Học máy cơ bản}
Học máy (Machine Learning - ML) là một lĩnh vực của Trí tuệ Nhân tạo (AI) cho phép các hệ thống máy tính tự động học hỏi và cải thiện từ kinh nghiệm mà không cần được lập trình rõ ràng. Trong chương này, chúng ta sẽ đi sâu vào việc ứng dụng và phân tích các kỹ thuật Học máy cơ bản thông qua ba bộ dữ liệu tiêu biểu, đại diện cho các bài toán phân loại (classification), hồi quy (regression) và xử lý ngôn ngữ tự nhiên cơ bản (NLP - text classification). Các thuật toán được triển khai bao gồm Logistic Regression, Linear Regression và Multilayer Perceptron (MLP) cho bài toán phân loại văn bản.

Đối với mỗi bộ dữ liệu, chúng ta sẽ khảo sát cấu trúc, phân phối dữ liệu, nền tảng lý thuyết toán học của mô hình học máy tương ứng, và sau đó triển khai mã nguồn dưới ba dạng: xây dựng mô hình từ đầu (From Scratch) bằng thư viện NumPy, sử dụng framework Keras (TensorFlow) và sử dụng framework PyTorch. Cuối cùng, chúng ta sẽ đánh giá và so sánh hiệu suất của các phương pháp này.

\subsubsection{Lý thuyết Hồi quy Logistic (Logistic Regression)}
Đối với bài toán phân loại nhị phân như dự đoán bệnh tiểu đường, mô hình Hồi quy Logistic (Logistic Regression) là một sự lựa chọn kinh điển và mạnh mẽ. 

\textbf{Mô hình dự đoán:}
Mục tiêu là mô hình hóa xác suất để $y=1$ cho trước vector đầu vào $\mathbf{x} = [x_1, x_2, ..., x_n]^T$. Chúng ta sử dụng hàm Sigmoid (Logistic function) để ánh xạ tổ hợp tuyến tính của các đặc trưng vào khoảng $(0, 1)$:
\begin{equation}
h_{\mathbf{\theta}}(\mathbf{x}) = \sigma(\mathbf{w}^T\mathbf{x} + b) = \frac{1}{1 + e^{-(\mathbf{w}^T\mathbf{x} + b)}}
\end{equation}
Trong đó:
\begin{itemize}
    \item $\mathbf{w}$ là vector trọng số (weights).
    \item $b$ là hệ số chệch (bias).
    \item $h_{\mathbf{\theta}}(\mathbf{x})$ đại diện cho xác suất $P(y=1|\mathbf{x}; \mathbf{w}, b)$.
\end{itemize}

\textbf{Hàm mất mát (Loss Function - Binary Cross Entropy):}
Để huấn luyện mô hình, chúng ta cần tối thiểu hóa hàm mất mát Cross Entropy:
\begin{equation}
J(\mathbf{w}, b) = -\frac{1}{m} \sum_{i=1}^{m} \left[ y^{(i)} \log(h_{\theta}(\mathbf{x}^{(i)})) + (1 - y^{(i)}) \log(1 - h_{\theta}(\mathbf{x}^{(i)})) \right]
\end{equation}
Trong đó $m$ là số lượng mẫu trong tập huấn luyện, $y^{(i)}$ là nhãn thực tế của mẫu thứ $i$.

\textbf{Cập nhật trọng số bằng Gradient Descent:}
\begin{align}
\frac{\partial J}{\partial \mathbf{w}} &= \frac{1}{m} \sum_{i=1}^{m} \left( h_{\theta}(\mathbf{x}^{(i)}) - y^{(i)} \right) \mathbf{x}^{(i)} \\
\frac{\partial J}{\partial b} &= \frac{1}{m} \sum_{i=1}^{m} \left( h_{\theta}(\mathbf{x}^{(i)}) - y^{(i)} \right) \\
\mathbf{w} &:= \mathbf{w} - \alpha \frac{\partial J}{\partial \mathbf{w}} \\
b &:= b - \alpha \frac{\partial J}{\partial b}
\end{align}
với $\alpha$ là tốc độ học (learning rate).

\subsubsection{Cài đặt mô hình}

\textbf{1. Cài đặt từ đầu (From Scratch) bằng NumPy}
\begin{lstlisting}[language=Python]
import numpy as np

class LogisticRegressionNumPy:
    def __init__(self, learning_rate=0.01, n_iters=1000):
        self.learning_rate = learning_rate
        self.n_iters = n_iters
        self.weights = None
        self.bias = None
        
    def _sigmoid(self, x):
        # Tranh overflow trong ham exp bang cach clip gia tri x
        x = np.clip(x, -250, 250)
        return 1 / (1 + np.exp(-x))
        
    def fit(self, X, y):
        n_samples, n_features = X.shape
        self.weights = np.zeros(n_features)
        self.bias = 0
        
        for _ in range(self.n_iters):
            linear_model = np.dot(X, self.weights) + self.bias
            y_predicted = self._sigmoid(linear_model)
            
            dw = (1 / n_samples) * np.dot(X.T, (y_predicted - y))
            db = (1 / n_samples) * np.sum(y_predicted - y)
            
            self.weights -= self.learning_rate * dw
            self.bias -= self.learning_rate * db
            
    def predict(self, X):
        linear_model = np.dot(X, self.weights) + self.bias
        y_predicted = self._sigmoid(linear_model)
        y_predicted_cls = [1 if i > 0.5 else 0 for i in y_predicted]
        return np.array(y_predicted_cls)
\end{lstlisting}

\subsubsection{Lý thuyết Hồi quy Tuyến tính Đa biến (Multiple Linear Regression)}
Để dự đoán giá nhà dựa trên nhiều đặc trưng đầu vào, Hồi quy Tuyến tính Đa biến là phương pháp cơ bản nhất. 

\textbf{Mô hình dự đoán:}
Giả sử ta có $n$ đặc trưng $\mathbf{x} = [x_1, x_2, ..., x_n]^T$, hàm giả thuyết (hypothesis) được định nghĩa là một tổ hợp tuyến tính của các đặc trưng:
\begin{equation}
h_{\mathbf{\theta}}(\mathbf{x}) = w_1 x_1 + w_2 x_2 + ... + w_n x_n + b = \mathbf{w}^T\mathbf{x} + b
\end{equation}

\textbf{Hàm mất mát (Mean Squared Error - MSE):}
Mục tiêu là tìm ra các tham số $\mathbf{w}$ và $b$ sao cho sai số bình phương trung bình giữa giá trị dự đoán và giá trị thực tế là nhỏ nhất:
\begin{equation}
J(\mathbf{w}, b) = \frac{1}{2m} \sum_{i=1}^{m} \left( h_{\theta}(\mathbf{x}^{(i)}) - y^{(i)} \right)^2
\end{equation}

\textbf{Cập nhật trọng số bằng Gradient Descent:}
\begin{align}
\frac{\partial J}{\partial \mathbf{w}} &= \frac{1}{m} \sum_{i=1}^{m} \left( h_{\theta}(\mathbf{x}^{(i)}) - y^{(i)} \right) \mathbf{x}^{(i)} \\
\frac{\partial J}{\partial b} &= \frac{1}{m} \sum_{i=1}^{m} \left( h_{\theta}(\mathbf{x}^{(i)}) - y^{(i)} \right)
\end{align}
Ngoài Gradient Descent, Hồi quy tuyến tính có thể được giải nghiệm đóng (closed-form solution) bằng phương trình Normal Equation: $\theta = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{y}$.

\begin{lstlisting}[language=Python]
import numpy as np

class LinearRegressionNumPy:
    def __init__(self, learning_rate=0.01, n_iters=1000):
        self.learning_rate = learning_rate
        self.n_iters = n_iters
        self.weights = None
        self.bias = None
        
    def fit(self, X, y):
        n_samples, n_features = X.shape
        self.weights = np.zeros(n_features)
        self.bias = 0
        
        for _ in range(self.n_iters):
            y_predicted = np.dot(X, self.weights) + self.bias
            
            dw = (1 / n_samples) * np.dot(X.T, (y_predicted - y))
            db = (1 / n_samples) * np.sum(y_predicted - y)
            
            self.weights -= self.learning_rate * dw
            self.bias -= self.learning_rate * db
            
    def predict(self, X):
        return np.dot(X, self.weights) + self.bias
        
    def mse(self, y_true, y_pred):
        return np.mean((y_true - y_pred)**2)

\subsubsection{Lý thuyết Mạng nơ-ron đa tầng (Multilayer Perceptron - MLP)}
Để giải quyết bài toán phân loại văn bản biểu diễn bởi TF-IDF vectors (thường có số chiều rất lớn), Mạng nơ-ron truyền thẳng (Feedforward Neural Network) hay Multilayer Perceptron (MLP) là mô hình lý tưởng.

\textbf{Mô hình dự đoán:}
Một mô hình MLP bao gồm một lớp đầu vào, một hoặc nhiều lớp ẩn (hidden layers) và một lớp đầu ra. Giả sử ta có 1 lớp ẩn, việc lan truyền tiến (forward propagation) diễn ra như sau:
\begin{align}
\mathbf{z}^{[1]} &= \mathbf{W}^{[1]}\mathbf{x} + \mathbf{b}^{[1]} \\
\mathbf{a}^{[1]} &= g^{[1]}(\mathbf{z}^{[1]}) = \text{ReLU}(\mathbf{z}^{[1]}) \\
\mathbf{z}^{[2]} &= \mathbf{W}^{[2]}\mathbf{a}^{[1]} + \mathbf{b}^{[2]} \\
\hat{y} &= \mathbf{a}^{[2]} = g^{[2]}(\mathbf{z}^{[2]}) = \sigma(\mathbf{z}^{[2]})
\end{align}
Trong đó $\mathbf{W}^{[l]}$ và $\mathbf{b}^{[l]}$ là ma trận trọng số và vector bias của lớp thứ $l$. Hàm kích hoạt ReLU được định nghĩa là $ReLU(z) = \max(0, z)$ nhằm giải quyết vấn đề triệt tiêu gradient (vanishing gradient). Hàm đầu ra là Sigmoid $\sigma(z) = \frac{1}{1 + e^{-z}}$ để cung cấp xác suất nhị phân.

\textbf{Hàm mất mát và Backpropagation:}
Hàm mất mát là Binary Cross Entropy. Quá trình học được thực hiện thông qua thuật toán Lan truyền ngược (Backpropagation) để tính toán gradient của Loss theo các tham số ở mọi lớp thông qua quy tắc chuỗi (Chain Rule) của giải tích.

\textbf{1. Cài đặt từ đầu (From Scratch) bằng NumPy}
\begin{lstlisting}[language=Python]
import numpy as np

class MLPNumPy:
    def __init__(self, input_size, hidden_size, output_size, lr=0.01):
        self.lr = lr
        # Khoi tao trong so (He initialization cho ReLU)
        self.W1 = np.random.randn(input_size, hidden_size) * np.sqrt(2. / input_size)
        self.b1 = np.zeros((1, hidden_size))
        self.W2 = np.random.randn(hidden_size, output_size) * np.sqrt(1. / hidden_size)
        self.b2 = np.zeros((1, output_size))
        
    def relu(self, Z):
        return np.maximum(0, Z)
        
    def relu_derivative(self, Z):
        return Z > 0
        
    def sigmoid(self, Z):
        return 1 / (1 + np.exp(-np.clip(Z, -250, 250)))
        
    def forward(self, X):
        self.Z1 = np.dot(X, self.W1) + self.b1
        self.A1 = self.relu(self.Z1)
        self.Z2 = np.dot(self.A1, self.W2) + self.b2
        self.A2 = self.sigmoid(self.Z2)
        return self.A2
        
    def backward(self, X, y):
        m = X.shape[0]
        y = y.reshape(-1, 1)
        
        dZ2 = self.A2 - y
        dW2 = (1/m) * np.dot(self.A1.T, dZ2)
        db2 = (1/m) * np.sum(dZ2, axis=0, keepdims=True)
        
        dA1 = np.dot(dZ2, self.W2.T)
        dZ1 = dA1 * self.relu_derivative(self.Z1)
        dW1 = (1/m) * np.dot(X.T, dZ1)
        db1 = (1/m) * np.sum(dZ1, axis=0, keepdims=True)
        
        self.W2 -= self.lr * dW2
        self.b2 -= self.lr * db2
        self.W1 -= self.lr * dW1
        self.b1 -= self.lr * db1

    def fit(self, X, y, epochs=1000):
        for epoch in range(epochs):
            self.forward(X)
            self.backward(X, y)
\end{lstlisting}
